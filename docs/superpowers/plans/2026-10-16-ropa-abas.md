# RoPA em abas, clone do legado (fase 1): plano de implementação

> **Para agentes:** SUB-SKILL OBRIGATÓRIA: use superpowers:subagent-driven-development (um worktree por pacote) ou superpowers:executing-plans. Os passos usam caixas `- [ ]`. Leia antes: `AGENTS.md`, `PRODUCT.md`, `design.md` e a spec [`2026-10-16-ropa-abas-design.md`](../specs/2026-10-16-ropa-abas-design.md) (as seções citadas como "spec 4.4" etc. são dela).

**Objetivo:** trocar a atividade de tratamento do n.privacy (entrevista A a H, página de cinco etapas e modal de Ligações) pela **ficha em abas** do legado: cabeçalho e nove abas livres (Escopo, Tratamento de Dados, Agentes de Tratamento, Medidas de Segurança, Base Jurídica, Controle de GAPs, Teste de Impacto, Riscos, Logs ROPA), com o ciclo de vida **por etapa**, a situação e o menu "Ações" mapeados sobre a aprovação por pedido (sem enfraquecê-la), e a entrevista guiada como segunda porta para o mesmo dado.

**Arquitetura:** migration **0079** aditiva (colunas novas em `ropa_records`, `tratamento_ciclo`, `tratamento_bases`, `corrective_actions`, `risks`; tabelas `catalogo_itens`, `tratamento_etapa_*`, `tratamento_interno_areas`, `tratamento_externo_parceiros`, `tratamento_medidas`, `tratamento_bases_gdpr`, `acao_anexos`; gatilhos de tenant). Nada novo entra no conteúdo congelado da aprovação (`src/services/pedidos.ts:125-142`, congelado): os dados de etapa **escrevem nas colunas que já estão no hash**, só quando a pessoa edita (spec 4.4). Rotas novas em `src/routes/ropa-abas.ts` (faixa `privacy`), serviços em `src/services/ropa-abas/`, esquemas zod em `src/schemas/ropa-abas.ts`. Frontend vanilla: a página `ropa-atividade` é reescrita como casca de abas, com um módulo por aba em `frontend/src/atividade/`.

**Stack:** Cloudflare Workers + Hono + D1, zod 4, vitest 4 (`@cloudflare/vitest-pool-workers` no backend, jsdom no frontend), Playwright.

import { lerCiclo, ETAPAS_CICLO } from '../ciclo';
import type { EstadoEtapa } from './tipos';

/**
 * Situação de cada etapa do Tratamento de Dados. PR1: lida do ciclo da F3 (`registrada` conta como concluída, `nao_aplicavel` como tal, sem
 * linha é pendente). PR2: o pacote A2 troca o corpo pela derivação por etapa (spec 5) sem mudar a assinatura.
 */
export async function estadoDasEtapas(db: D1Database, projectId: string, ropaId: string): Promise<EstadoEtapa[]> {
  const ciclo = (await lerCiclo(db, projectId, ropaId)) ?? [];
  return ETAPAS_CICLO.map((e) => {
    const l = ciclo.find((x) => x.etapa === e.id);
    const situacao = l?.situacao === 'registrada' ? 'concluida' : l?.situacao === 'nao_aplicavel' ? 'nao_aplicavel' : 'pendente';
    return { etapa: e.id, situacao, faltas: situacao === 'pendente' ? ['etapa ainda não respondida'] : [] };
  });
}

/** Seções legíveis da trilha, como o legado as escreve ("itens: Controle de GAPS / Alteração"). */
export const SECOES = ['Descrição da atividade', 'Escopo', 'Tratamento de Dados', 'Agentes de Tratamento', 'Medidas de Segurança', 'Base Jurídica',
  'Controle de GAPS', 'Teste de Impacto', 'Riscos', 'Situação'] as const;
export type Secao = (typeof SECOES)[number];

/** Ação antiga (F1 a F3) → seção, para o histórico que as rotas de antes já gravaram. */
const SECAO_DA_ACAO: Record<string, Secao> = {
  'ropa.impacto': 'Teste de Impacto', 'ropa.ciclo': 'Tratamento de Dados', 'ropa.itens': 'Tratamento de Dados', 'ropa.departamentos': 'Escopo',
  'ropa.bases': 'Base Jurídica', 'ropa_updated': 'Escopo', 'ropa_created': 'Escopo', 'ropa.approval_revoked': 'Situação',
};

/**
 * Uma linha de trilha por seção alterada. SÓ seção, ação e ids: texto livre digitado pela pessoa (descrição, gap, medida) nunca entra aqui.
 * `detalhe` é opcional e curto (contagem ou código, não texto).
 */
export async function registrarSecao(db: D1Database, ator: string, projectId: string, ropaId: string, secao: Secao | string, evento: string, detalhe = '') {
  await db.prepare(
    `INSERT INTO audit_logs (id, action, actor, details, project_id, entity_type, entity_id, field, created_at)
     VALUES (?, 'ropa.secao', ?, ?, ?, 'ropa', ?, ?, datetime('now'))`
  ).bind(genId(), ator, `ROPA ${ropaId}: ${evento}${detalhe ? ` (${detalhe})` : ''}`, projectId, ropaId, `${secao} / ${evento}`).run();
}

export type LinhaDeLog = { secao: string; evento: string; nome: string; data: string };

export async function lerLogs(db: D1Database, projectId: string, ropaId: string, { limite = 50, antes }: { limite?: number; antes?: string } = {}): Promise<LinhaDeLog[] | null> {
  if (!(await db.prepare('SELECT 1 FROM ropa_records WHERE id = ? AND project_id = ?').bind(ropaId, projectId).first())) return null;
  const { results } = await db.prepare(
    `SELECT l.action, l.actor, l.field, l.created_at, COALESCE(NULLIF(u.name, ''), l.actor) AS nome
       FROM audit_logs l LEFT JOIN users u ON lower(u.email) = lower(l.actor)
      WHERE l.project_id = ?1 AND ((l.entity_type = 'ropa' AND l.entity_id = ?2) OR l.details LIKE ?3)
        AND (?4 IS NULL OR l.created_at < ?4)
      ORDER BY l.created_at DESC, l.rowid DESC LIMIT ?5`
  ).bind(projectId, ropaId, `ROPA ${ropaId}:%`, antes ?? null, Math.min(Math.max(limite, 1), 200))
    .all<{ action: string; actor: string; field: string | null; created_at: string; nome: string }>();
  return results.map((r) => {
    const [secao, evento] = r.field ? r.field.split(' / ') : [SECAO_DA_ACAO[r.action] ?? 'Escopo', r.action.replace(/^ropa[._]/, '').replace(/_/g, ' ')];
    return { secao, evento: evento ?? 'Alteração', nome: r.nome, data: r.created_at };
  });
}

import { genId, sha256Hex } from '../../helpers';
import { falha, type Falha } from './comum';

export const MAX_ANEXO = 5 * 1024 * 1024;
type Tipo = 'pdf' | 'doc' | 'docx';
const MAGIA: Record<Tipo, number[]> = { pdf: [0x25, 0x50, 0x44, 0x46], doc: [0xd0, 0xcf, 0x11, 0xe0], docx: [0x50, 0x4b, 0x03, 0x04] };

/** O tipo pelo CONTEÚDO (os primeiros bytes), conferido com a extensão: renomear um executável para .pdf não passa. */
export function tipoDoAnexo(nome: string, bytes: Uint8Array): Tipo | null {
  const ext = nome.toLowerCase().split('.').pop() as Tipo;
  const magia = MAGIA[ext];
  if (!magia) return null;
  return magia.every((b, i) => bytes[i] === b) ? ext : null;
}

export async function guardarAnexo(env: { DB: D1Database; STORAGE: R2Bucket }, projectId: string, ropaId: string, acaoId: string, ator: string, arq: { name: string; bytes: ArrayBuffer }): Promise<Falha | { ok: true; id: string }> {
  if (arq.bytes.byteLength > MAX_ANEXO) return falha(400, 'Máximo de 5 MB por arquivo');
  const tipo = tipoDoAnexo(arq.name, new Uint8Array(arq.bytes.slice(0, 8)));
  if (!tipo) return falha(400, 'Envie um arquivo PDF ou DOC (máx. 5MB) válido');
  const acao = await env.DB.prepare('SELECT 1 FROM corrective_actions WHERE id = ? AND ropa_id = ? AND project_id = ?').bind(acaoId, ropaId, projectId).first();
  if (!acao) return falha(404, 'Gap não encontrado');
  const id = genId();
  const chave = `acoes/${projectId}/${acaoId}/${id}`;
  await env.STORAGE.put(chave, arq.bytes, { httpMetadata: { contentType: tipo === 'pdf' ? 'application/pdf' : tipo === 'doc' ? 'application/msword' : 'application/vnd.openxmlformats-officedocument.wordprocessingml.document' } });
  await env.DB.prepare('INSERT INTO acao_anexos (id, project_id, acao_id, file_name, r2_key, file_hash, file_type, file_size, uploaded_by) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)')
    .bind(id, projectId, acaoId, arq.name.slice(0, 255), chave, await sha256Hex(new Uint8Array(arq.bytes)), tipo, arq.bytes.byteLength, ator).run();
  return { ok: true, id };
}
